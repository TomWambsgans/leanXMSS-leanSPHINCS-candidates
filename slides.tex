% --- a tree with no root: levels 1..#4 (2,4,8,... nodes), path to leaf #7 ---
% #1 name  #2 centre x  #3 y of the top level  #4 height  #5 leaf sp  #6 level sp  #7 leaf
\newcommand{\fstree}[7]{%
  \foreach \l in {1,...,#4} {
    \pgfmathtruncatemacro{\n}{2^\l-1}
    \pgfmathtruncatemacro{\p}{floor(#7/2^(#4-\l))}
    \pgfmathtruncatemacro{\q}{\p+1-2*mod(\p,2)}
    \foreach \j in {0,...,\n} {
      \pgfmathsetmacro{\x}{#2+((\j+0.5)*2^(#4-\l)-2^(#4-1))*#5}
      \pgfmathsetmacro{\y}{#3-(\l-1)*#6}
      \ifnum\j=\p \node[pathnode] (#1-\l-\j) at (\x,\y) {};
      \else\ifnum\j=\q \node[authnode] (#1-\l-\j) at (\x,\y) {};
      \else \node[plain] (#1-\l-\j) at (\x,\y) {};\fi\fi}}
  \foreach \l in {2,...,#4} {
    \pgfmathtruncatemacro{\n}{2^\l-1}
    \pgfmathtruncatemacro{\p}{floor(#7/2^(#4-\l))}
    \pgfmathtruncatemacro{\u}{\l-1}
    \foreach \j in {0,...,\n} {
      \pgfmathtruncatemacro{\k}{floor(\j/2)}
      \ifnum\j=\p \draw[hot] (#1-\u-\k) -- (#1-\l-\j);
      \else \draw[edge] (#1-\u-\k) -- (#1-\l-\j);\fi}}}

\begin{itemize}\setlength{\itemsep}{12pt}
  \item \textbf{novel}.
  \item \textbf{Security in the standard model will require a new property of the hash (in the same way ``interleaved target subset resilience'' has been created ad-hoc for FORS)}.

  \item \textbf{Higher variance of the security degradation}.
\end{itemize}

%============================================== 4. the semi-trusted state ===
\begin{frame}{Partial fix: a (semi-trusted) state of the leaves already signed}
\vspace{0.4em}
\begin{itemize}\setlength{\itemsep}{14pt}
  \item Keep track of $S$, the leaves already signed. When grinding ends up on $S$ again, accept only with
        probability $\frac{2^b-|S|}{2^h-|S|}$, else keep grinding. \textbf{Only while $|S| < 2^{b-1}$.}
  \item \textbf{Zero leak} up to $2^{b-1}$: 2\,048 signatures on the Ledger, with $b=12$.
  \item Price: up to $2\times$ more grinding, and $2\times$ less lifetime if you do not trust $S$.
  \item An adversary controlling $S$ can:
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item undo the privacy guarantee: $b$ leaks again, with enough signatures;
      \item divide the lifetime by 2.
    \end{itemize}
    It can never forge.
\end{itemize}
\end{frame}
